Cloud Firestore, Sheet & GitHub

Koneksi otomatis lintas perangkat & penyimpanan aman untuk GitHub

Status Koneksi Firestore: Terhubung

Project "pitabsensi" aktif & tersinkronisasi otomatis.

Tempel Kode `firebaseConfig`

Wajib untuk GitHub: Tanamkan Config ke `index.html`

Agar semua perangkat yang membuka link GitHub Anda otomatis terhubung ke Firestore tanpa mengatur ulang, klik tombol di bawah lalu upload file index.html hasil unduhan ke repository GitHub Anda.

Lihat Firestore Security Rules (Salin ke Firebase) 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /workforce_pro/{docId} {
      allow read, write: if true;
    }
  }
}

Pastikan Anda sudah klik Firestore Database → Create database di Firebase Console dan menempelkan Rules di atas pada tab Rules.

Sistem Anti-Double File & Anti-Double Baris

Data absensi otomatis dikirim ke 1 file Google Spreadsheet tetap menggunakan kunci unik Tanggal + ID Karyawan. Jika absensi diedit, baris lama ditimpa (tidak membuat file/baris ganda).

Salin Kode Apps Script Anti-Double (Code.gs)

Buka Google Spreadsheet Anda → Ekstensi → Apps Script → tempel kode ini lalu Deploy → Web App (Anyone):

const SECRET_TOKEN = "GANTI_DENGAN_TOKEN_RAHASIA_ANDA";

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const body = JSON.parse(e.postData.contents);
    if (SECRET_TOKEN && body.token !== SECRET_TOKEN) {
      return ContentService.createTextOutput(JSON.stringify({ ok: false, error: "Unauthorized" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const state = body.fullState || {};

    // 1. Sheet Absensi (Anti-Double berbasis Unique_Key: Tanggal__ID_Karyawan)
    const shAbs = getOrCreateSheet(ss, "Absensi", [
      "Unique_Key", "ID_Absensi", "Tanggal", "ID_Karyawan", "Nama_Karyawan",
      "Status_Hadir", "Jam_Lembur", "Upah_Harian", "Upah_Lembur", "Total_Upah", "Catatan", "Diperbarui_Pada"
    ]);
    upsertAbsensiRows(shAbs, state.absensi || [], body.action, body.payload);

    // 2. Sheet Karyawan
    const shEmp = getOrCreateSheet(ss, "Karyawan", [
      "ID_Karyawan", "Nama_Lengkap", "No_HP", "Alamat", "Gaji_Harian", "Tarif_Lembur_Jam"
    ]);
    syncKaryawanSheet(shEmp, state.karyawan || []);

    // 3. Sheet Kasbon
    const shKas = getOrCreateSheet(ss, "Kasbon", [
      "ID_Kasbon", "Tanggal", "ID_Karyawan", "Nama_Karyawan", "Nominal", "Keterangan", "Status"
    ]);
    syncKasbonSheet(shKas, state.kasbon || []);

    return ContentService.createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: String(err) }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function getOrCreateSheet(ss, name, headers) {
  let sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    sh.appendRow(headers);
    sh.getRange(1, 1, 1, headers.length).setFontWeight("bold");
  }
  return sh;
}

function upsertAbsensiRows(sh, allAbsensi, action, payload) {
  if (action === "DELETE_ABSENSI" && payload && payload.id) {
    const lastRow = sh.getLastRow();
    if (lastRow > 1) {
      const vals = sh.getRange(2, 1, lastRow - 1, 2).getValues();
      for (let i = vals.length - 1; i >= 0; i--) {
        if (String(vals[i][1]) === String(payload.id)) {
          sh.deleteRow(i + 2);
        }
      }
    }
    return;
  }

  const lastRow = sh.getLastRow();
  const keyToRow = {};
  if (lastRow > 1) {
    const existingKeys = sh.getRange(2, 1, lastRow - 1, 1).getValues();
    existingKeys.forEach((r, idx) => {
      if (r[0]) keyToRow[String(r[0])] = idx + 2;
    });
  }

  const nowStr = Utilities.formatDate(new Date(), "Asia/Makassar", "yyyy-MM-dd HH:mm:ss");
  allAbsensi.forEach(a => {
    const uniqueKey = a.tanggal + "__" + a.karyawanId;
    const rowData = [
      uniqueKey, a.id, a.tanggal, a.karyawanId, a.namaKaryawan,
      a.tipe, Number(a.jamLembur || 0), Number(a.upahHarian || 0),
      Number(a.upahLembur || 0), Number(a.totalUpah || 0), a.catatan || "", nowStr
    ];
    if (keyToRow[uniqueKey]) {
      sh.getRange(keyToRow[uniqueKey], 1, 1, rowData.length).setValues([rowData]);
    } else {
      sh.appendRow(rowData);
      keyToRow[uniqueKey] = sh.getLastRow();
    }
  });
}

function syncKaryawanSheet(sh, list) {
  const lastRow = sh.getLastRow();
  if (lastRow > 1) sh.getRange(2, 1, lastRow - 1, 6).clearContent();
  if (list.length === 0) return;
  const rows = list.map(k => [k.id, k.nama, k.hp, k.alamat, Number(k.gajiHarian || 0), Number(k.lemburJam || 0)]);
  sh.getRange(2, 1, rows.length, 6).setValues(rows);
}

function syncKasbonSheet(sh, list) {
  const lastRow = sh.getLastRow();
  if (lastRow > 1) sh.getRange(2, 1, lastRow - 1, 7).clearContent();
  if (list.length === 0) return;
  const rows = list.map(k => [k.id, k.tanggal, k.karyawanId, k.namaKaryawan, Number(k.nominal || 0), k.keterangan, k.status]);
  sh.getRange(2, 1, rows.length, 7).setValues(rows);
}
