Sistem Anti-Double Spreadsheet & Token Aman

URL & Token Spreadsheet disimpan di Cloud Firestore (bukan di kode GitHub), sehingga otomatis aktif di semua HP setelah login tanpa membocorkan token di repositori publik.

Kode Google Apps Script (Anti-Double File & Baris) 
/**
 * GOOGLE APPS SCRIPT - WORKFORCE PRO (ANTI-DOUBLE FILE & ANTI-DOUBLE BARIS)
 * Cara Pasang:
 * 1. Buka Google Spreadsheet Anda -> Menu Ekstensi (Extensions) -> Apps Script.
 * 2. Hapus kode lama, tempel seluruh kode ini, lalu ganti SECRET_TOKEN sesuai pengaturan di aplikasi.
 * 3. Klik Deploy -> New deployment -> Web app -> Execute as: Me -> Who has access: Anyone.
 * 4. Salin Web App URL dan tempel di menu Cloud -> 2. Spreadsheet pada aplikasi.
 */
const SECRET_TOKEN = "GANTI_DENGAN_TOKEN_RAHASIA_ANDA";
const SHEET_NAME = "Absensi_Karyawan";

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(15000); // Mencegah bentrok & duplikasi saat input bersamaan

  try {
    const body = JSON.parse(e.postData.contents);
    if (SECRET_TOKEN && body.token !== SECRET_TOKEN) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Token tidak valid" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // Menggunakan Spreadsheet aktif saat ini (TIDAK membuat file baru / double)
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) {
      sheet = ss.insertSheet(SHEET_NAME);
      sheet.appendRow([
        "Unique_Key", "Tanggal", "ID_Karyawan", "Nama_Karyawan",
        "Status_Hadir", "Jam_Lembur", "Upah_Harian", "Upah_Lembur",
        "Total_Upah", "Catatan", "Input_Oleh", "Waktu_Update"
      ]);
      sheet.getRange(1, 1, 1, 12).setFontWeight("bold").setBackground("#e2e8f0");
      sheet.setFrozenRows(1);
    }

    const action = body.action;
    const payload = body.payload;

    if (action === "UPSERT_ABSENSI") {
      const records = Array.isArray(payload) ? payload : [payload];
      upsertRecords(sheet, records);
    } else if (action === "DELETE_ABSENSI") {
      const targetKey = payload && payload.uniqueKey ? payload.uniqueKey : "";
      if (targetKey) deleteRecordByKey(sheet, targetKey);
    } else if (action === "SYNC_ALL_ABSENSI") {
      const records = Array.isArray(payload) ? payload : [];
      const lastRow = sheet.getLastRow();
      if (lastRow > 1) {
        sheet.getRange(2, 1, lastRow - 1, 12).clearContent();
      }
      if (records.length > 0) {
        const rows = records.map(r => formatRow(r));
        sheet.getRange(2, 1, rows.length, 12).setValues(rows);
      }
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function formatRow(r) {
  const uKey = r.uniqueKey || (r.tanggal + "__" + r.karyawanId);
  return [
    uKey,
    r.tanggal || "",
    r.karyawanId || "",
    r.namaKaryawan || "",
    r.tipe || "",
    Number(r.jamLembur || 0),
    Number(r.upahHarian || 0),
    Number(r.upahLembur || 0),
    Number(r.totalUpah || 0),
    r.catatan || "",
    r.inputOleh || "",
    new Date().toLocaleString("id-ID")
  ];
}

function upsertRecords(sheet, records) {
  const lastRow = sheet.getLastRow();
  const existingKeys = lastRow > 1
    ? sheet.getRange(2, 1, lastRow - 1, 1).getValues().map(row => String(row[0]))
    : [];

  records.forEach(r => {
    const rowData = formatRow(r);
    const uKey = String(rowData[0]);
    const foundIndex = existingKeys.indexOf(uKey);
    if (foundIndex >= 0) {
      // Timpa baris yang sudah ada agar TIDAK DOUBLE
      sheet.getRange(foundIndex + 2, 1, 1, 12).setValues([rowData]);
    } else {
      sheet.appendRow(rowData);
      existingKeys.push(uKey);
    }
  });
}

function deleteRecordByKey(sheet, uniqueKey) {
  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return;
  const keys = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
  for (let i = keys.length - 1; i >= 0; i--) {
    if (String(keys[i][0]) === String(uniqueKey)) {
      sheet.deleteRow(i + 2);
    }
  }
}

Ubah PIN Admin & Mandor (Enkripsi SHA-256)

Kosongkan kolom jika tidak ingin mengubah PIN saat ini. PIN disimpan dalam bentuk hash SHA-256 di Cloud Firestore.

Firestore Security Rules (Wajib Aktifkan Auth Anonymous) 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /artifacts/{appId}/public/data/{document=**} {
      allow read, write: if request.auth != null;
    }
  }
}

Tips Keamanan GitHub: Aktifkan Anonymous Sign-In di Firebase Authentication, dan batasi HTTP Referrer API Key Anda di Google Cloud Console ke domain https://username.github.io/*.
