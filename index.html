Keamanan & Anti-Double Spreadsheet:

URL & Token Spreadsheet disimpan di Cloud Firestore (tidak tertulis di file GitHub). Setiap absensi memakai Unique_Key (Tanggal__ID) sehingga mengedit/menyimpan ulang akan menimpa baris lama, bukan membuat baris atau file ganda.

Script Google Sheets Anti-Double (`Code.gs`) 
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    var body = JSON.parse(e.postData.contents);
    var props = PropertiesService.getScriptProperties();
    var savedToken = props.getProperty('SHEET_TOKEN') || 'RAHASIA_PIT_2026';
    if (body.token !== savedToken) {
      return ContentService.createTextOutput(JSON.stringify({ ok: false, error: 'Invalid token' })).setMimeType(ContentService.MimeType.JSON);
    }

    // Menggunakan Spreadsheet yang sedang aktif (TIDAK membuat file baru/double)
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('Absensi_PIT') || ss.insertSheet('Absensi_PIT');
    var headers = ['Unique_Key', 'Tanggal', 'ID_Karyawan', 'Nama_Karyawan', 'Status_Hadir', 'Jam_Lembur', 'Upah_Harian', 'Upah_Lembur', 'Total_Upah', 'Catatan', 'Input_Oleh', 'Updated_At'];

    if (sheet.getLastRow() === 0) {
      sheet.appendRow(headers);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold');
    }

    var records = body.records || [];
    if (records.length === 0) {
      return ContentService.createTextOutput(JSON.stringify({ ok: true, count: 0 })).setMimeType(ContentService.MimeType.JSON);
    }

    var lastRow = sheet.getLastRow();
    var keyMap = {};
    if (lastRow > 1) {
      var existingKeys = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
      for (var i = 0; i < existingKeys.length; i++) {
        var k = String(existingKeys[i][0]).trim();
        if (k) keyMap[k] = i + 2;
      }
    }

    var nowStr = new Date().toLocaleString('id-ID');
    records.forEach(function(r) {
      var uKey = r.uniqueKey || (r.tanggal + '__' + r.karyawanId);
      var rowValues = [
        uKey,
        r.tanggal,
        r.karyawanId,
        r.namaKaryawan,
        r.tipe,
        Number(r.jamLembur || 0),
        Number(r.upahHarian || 0),
        Number(r.upahLembur || 0),
        Number(r.totalUpah || 0),
        r.catatan || '',
        r.inputOleh || '',
        nowStr
      ];
      if (keyMap[uKey]) {
        // Timpa baris yang sudah ada (Anti-Double)
        sheet.getRange(keyMap[uKey], 1, 1, rowValues.length).setValues([rowValues]);
      } else {
        sheet.appendRow(rowValues);
        keyMap[uKey] = sheet.getLastRow();
      }
    });

    return ContentService.createTextOutput(JSON.stringify({ ok: true, synced: records.length })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
