URL ini otomatis ikut tersimpan di Firestore sehingga HP lain otomatis kirim ke Spreadsheet yang sama.

Solusi Ganti HP / Perangkat Tanpa Setup Ulang:

Klik "Unduh File App Terkonfigurasi (.html)" di bawah untuk mendapatkan file aplikasi yang sudah tertanam permanen dengan API Key, Project ID Firestore & URL Spreadsheet Anda, atau bagikan Link Auto-Connect.

Kode Google Apps Script (Anti-Double File & Data)

Buka Google Spreadsheet Anda → menu Ekstensi → Apps Script, tempel kode di bawah ini lalu klik Deploy → New Deployment (Web App, Who has access: Anyone). Kode ini memakai sistem Single-Spreadsheet & Upsert sehingga file dan baris absensi dijamin TIDAK AKAN DOUBLE: 

const SECRET_TOKEN = "RAHASIA_PROYEK_2026";
const SPREADSHEET_NAME = "Database_Workforce_Pro";

function getOrCreateSingleSpreadsheet() {
  try {
    const active = SpreadsheetApp.getActiveSpreadsheet();
    if (active) return active;
  } catch (e) {}

  const props = PropertiesService.getScriptProperties();
  const savedId = props.getProperty("SINGLE_SHEET_ID");
  if (savedId) {
    try { return SpreadsheetApp.openById(savedId); } catch (e) {}
  }

  const files = DriveApp.getFilesByName(SPREADSHEET_NAME);
  if (files.hasNext()) {
    const file = files.next();
    props.setProperty("SINGLE_SHEET_ID", file.getId());
    return SpreadsheetApp.openById(file.getId());
  }

  const ss = SpreadsheetApp.create(SPREADSHEET_NAME);
  props.setProperty("SINGLE_SHEET_ID", ss.getId());
  return ss;
}

function getOrInitSheet(ss, name, headers) {
  let sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  if (sh.getLastRow() === 0) {
    sh.appendRow(headers);
    sh.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#e2e8f0");
  }
  return sh;
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const body = JSON.parse(e.postData.contents || "{}");
    if (SECRET_TOKEN && body.token !== SECRET_TOKEN) {
      return ContentService.createTextOutput(JSON.stringify({ status: "UNAUTHORIZED" }));
    }

    const ss = getOrCreateSingleSpreadsheet();
    const fullState = body.fullState || {};

    const shAbs = getOrInitSheet(ss, "Data_Absensi", [
      "Kunci_Unik", "ID_Absensi", "Tanggal", "ID_Karyawan", "Nama_Karyawan",
      "Status_Hadir", "Jam_Lembur", "Upah_Harian", "Upah_Lembur", "Total_Upah", "Lokasi_Catatan", "Diinput_Oleh", "Waktu_Sync"
    ]);
    if (Array.isArray(fullState.absensi)) {
      const uniqueMap = {};
      fullState.absensi.forEach(a => {
        const key = (a.tanggal || "") + "_" + (a.karyawanId || "");
        uniqueMap[key] = a;
      });
      const sortedAbs = Object.values(uniqueMap).sort((a, b) => (b.tanggal || "").localeCompare(a.tanggal || ""));
      if (shAbs.getLastRow() > 1) {
        shAbs.getRange(2, 1, shAbs.getLastRow() - 1, 13).clearContent();
      }
      if (sortedAbs.length > 0) {
        const rows = sortedAbs.map(a => [
          (a.tanggal || "") + "_" + (a.karyawanId || ""),
          a.id || "", a.tanggal || "", a.karyawanId || "", a.namaKaryawan || "",
          a.tipe || "FULL", Number(a.jamLembur || 0), Number(a.upahHarian || 0),
          Number(a.upahLembur || 0), Number(a.totalUpah || 0), a.catatan || "",
          a.inputOleh || "Sistem", new Date().toLocaleString("id-ID")
        ]);
        shAbs.getRange(2, 1, rows.length, 13).setValues(rows);
      }
    }

    const shEmp = getOrInitSheet(ss, "Data_Karyawan", [
      "ID_Karyawan", "Nama_Lengkap", "No_HP", "Alamat", "Gaji_Harian", "Lembur_Per_Jam"
    ]);
    if (Array.isArray(fullState.karyawan)) {
      if (shEmp.getLastRow() > 1) shEmp.getRange(2, 1, shEmp.getLastRow() - 1, 6).clearContent();
      if (fullState.karyawan.length > 0) {
        const empRows = fullState.karyawan.map(k => [
          k.id || "", k.nama || "", k.hp || "", k.alamat || "", Number(k.gajiHarian || 0), Number(k.lemburJam || 0)
        ]);
        shEmp.getRange(2, 1, empRows.length, 6).setValues(empRows);
      }
    }

    const shKas = getOrInitSheet(ss, "Data_Kasbon", [
      "ID_Kasbon", "Tanggal", "ID_Karyawan", "Nama_Karyawan", "Nominal", "Keterangan", "Status"
    ]);
    if (Array.isArray(fullState.kasbon)) {
      if (shKas.getLastRow() > 1) shKas.getRange(2, 1, shKas.getLastRow() - 1, 7).clearContent();
      if (fullState.kasbon.length > 0) {
        const kasRows = fullState.kasbon.map(k => [
          k.id || "", k.tanggal || "", k.karyawanId || "", k.namaKaryawan || "",
          Number(k.nominal || 0), k.keterangan || "", k.status || "BELUM_LUNAS"
        ]);
        shKas.getRange(2, 1, kasRows.length, 7).setValues(kasRows);
      }
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "OK", syncedAt: new Date().toISOString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
