Kode Apps Script (Foto Drive Hari & Tanggal + Anti-Double)

Buka Google Spreadsheet Anda → menu Ekstensi → Apps Script, tempel kode terbaru di bawah ini lalu klik Deploy → New Deployment (Web App, Who has access: Anyone). Semua foto absensi otomatis tersimpan di Google Drive folder Foto_Absensi_PIT_Absensi / [Hari, Tanggal] dengan nama file [Hari, Tanggal] - [Nama Karyawan].jpg: 

const SECRET_TOKEN = "RAHASIA_PROYEK_2026";
const SPREADSHEET_NAME = "Database_PIT_Absensi";
const DRIVE_FOLDER_NAME = "Foto_Absensi_PIT_Absensi";

function formatHariTanggalIndo(tanggalStr) {
  const hariList = ["Minggu", "Senin", "Selasa", "Rabu", "Kamis", "Jumat", "Sabtu"];
  const parts = String(tanggalStr || "").split("-");
  if (parts.length !== 3) return String(tanggalStr || "Tanggal");
  const y = Number(parts[0]);
  const m = Number(parts[1]);
  const d = Number(parts[2]);
  const dt = new Date(y, m - 1, d);
  const namaHari = hariList[dt.getDay()] || "";
  const dd = String(d).padStart(2, "0");
  const mm = String(m).padStart(2, "0");
  return namaHari + ", " + dd + "-" + mm + "-" + y;
}

function buildDriveFileName(tanggalStr, namaKaryawan) {
  const hariTanggal = formatHariTanggalIndo(tanggalStr);
  const cleanEmp = String(namaKaryawan || "Karyawan").replace(/[\\/:*?"<>|]/g, "").trim();
  return hariTanggal + " - " + cleanEmp + ".jpg";
}

function getOrCreateSingleSpreadsheet() {
  try {
    const active = SpreadsheetApp.getActiveSpreadsheet();
    if (active) return active;
  } catch (e) {}

  const props = PropertiesService.getScriptProperties();
  const savedId = props.getProperty("SINGLE_SHEET_ID");
  if (savedId) {
    try {
      return SpreadsheetApp.openById(savedId);
    } catch (e) {}
  }

  const files = DriveApp.getFilesByName(SPREADSHEET_NAME);
  if (files.hasNext()) {
    const existing = SpreadsheetApp.open(files.next());
    props.setProperty("SINGLE_SHEET_ID", existing.getId());
    return existing;
  }

  const created = SpreadsheetApp.create(SPREADSHEET_NAME);
  props.setProperty("SINGLE_SHEET_ID", created.getId());
  return created;
}

function getOrCreateDrivePhotoFolder(subFolderName) {
  const props = PropertiesService.getScriptProperties();
  let rootFolder = null;
  const savedFolderId = props.getProperty("PHOTO_FOLDER_ID");
  if (savedFolderId) {
    try {
      const f = DriveApp.getFolderById(savedFolderId);
      if (!f.isTrashed()) rootFolder = f;
    } catch (e) {}
  }

  if (!rootFolder) {
    const folders = DriveApp.getFoldersByName(DRIVE_FOLDER_NAME);
    if (folders.hasNext()) {
      rootFolder = folders.next();
    } else {
      rootFolder = DriveApp.createFolder(DRIVE_FOLDER_NAME);
    }
    props.setProperty("PHOTO_FOLDER_ID", rootFolder.getId());
  }

  if (!subFolderName) return rootFolder;
  const cleanSub = String(subFolderName).replace(/[\\/:*?"<>|]/g, "").trim();
  const subFolders = rootFolder.getFoldersByName(cleanSub);
  if (subFolders.hasNext()) {
    return subFolders.next();
  }
  return rootFolder.createFolder(cleanSub);
}

function saveBase64PhotoToDrive(base64DataUrl, fileName, subFolderName) {
  if (!base64DataUrl || String(base64DataUrl).indexOf("data:image") !== 0) {
    return null;
  }
  const parts = String(base64DataUrl).split(",");
  const mimeMatch = parts[0].match(/:(.*?);/);
  const mimeType = mimeMatch ? mimeMatch[1] : "image/jpeg";
  const bytes = Utilities.base64Decode(parts[1]);
  const blob = Utilities.newBlob(bytes, mimeType, fileName);

  const folder = getOrCreateDrivePhotoFolder(subFolderName);
  const oldFiles = folder.getFilesByName(fileName);
  while (oldFiles.hasNext()) {
    oldFiles.next().setTrashed(true);
  }

  const file = folder.createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Access.VIEW);
  const fileId = file.getId();

  return {
    fileId: fileId,
    fileName: fileName,
    folderName: folder.getName(),
    photoUrl: "https://drive.google.com/thumbnail?id=" + fileId + "&sz=w800",
    viewUrl: "https://drive.google.com/file/d/" + fileId + "/view"
  };
}

function upsertRowsByKey(sheet, headers, rows, keyIndex) {
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
  }
  const lastRow = sheet.getLastRow();
  const existingMap = {};
  if (lastRow > 1) {
    const existingKeys = sheet.getRange(2, keyIndex + 1, lastRow - 1, 1).getValues();
    for (let i = 0; i < existingKeys.length; i++) {
      const k = String(existingKeys[i][0] || "").trim();
      if (k) existingMap[k] = i + 2;
    }
  }
  rows.forEach(function(row) {
    const key = String(row[keyIndex] || "").trim();
    if (key && existingMap[key]) {
      sheet.getRange(existingMap[key], 1, 1, row.length).setValues([row]);
    } else {
      sheet.appendRow(row);
      if (key) existingMap[key] = sheet.getLastRow();
    }
  });
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const data = JSON.parse(e.postData.contents);
    if (SECRET_TOKEN && data.token !== SECRET_TOKEN) {
      return ContentService.createTextOutput(JSON.stringify({ ok: false, error: "Invalid token" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (data.action === "UPLOAD_SELFIE_DRIVE" && data.payload) {
      const p = data.payload;
      const hariTanggal = p.hariTanggal || formatHariTanggalIndo(p.tanggal);
      const fileName = p.fileName || buildDriveFileName(p.tanggal, p.namaKaryawan || p.karyawanId);
      const uploaded = saveBase64PhotoToDrive(p.base64Image, fileName, hariTanggal);
      if (!uploaded) {
        return ContentService.createTextOutput(JSON.stringify({ ok: false, error: "Format foto tidak valid" }))
          .setMimeType(ContentService.MimeType.JSON);
      }
      return ContentService.createTextOutput(JSON.stringify({
        ok: true,
        fileId: uploaded.fileId,
        fileName: uploaded.fileName,
        folderName: uploaded.folderName,
        photoUrl: uploaded.photoUrl,
        viewUrl: uploaded.viewUrl
      })).setMimeType(ContentService.MimeType.JSON);
    }

    const ss = getOrCreateSingleSpreadsheet();
    const fullState = data.fullState || {};
    const payloadObj = data.payload || {};

    if (Array.isArray(fullState.absensi)) {
      const shAbs = ss.getSheetByName("Absensi") || ss.insertSheet("Absensi");
      const headers = ["ID_Absensi", "Hari_Tanggal", "Tanggal", "Waktu_Absen", "ID_Karyawan", "Nama_Karyawan", "Jenis_Pekerjaan", "Tipe_Hadir", "Jam_Lembur", "Upah_Harian", "Upah_Lembur", "Total_Upah", "Lokasi_GPS", "Nama_File_Foto_Drive", "Link_Foto_Drive", "Catatan", "Input_Oleh"];
      const rows = fullState.absensi.map(function(a) {
        const hariTanggal = formatHariTanggalIndo(a.tanggal);
        const fileName = a.fotoFileName || buildDriveFileName(a.tanggal, a.namaKaryawan || a.karyawanId);
        let driveLink = a.fotoDriveUrl || (a.fotoSelfie && String(a.fotoSelfie).indexOf("http") === 0 ? a.fotoSelfie : "");

        const rawBase64 = a.fotoBase64 || (payloadObj && payloadObj.id === a.id ? payloadObj.fotoBase64 : "");
        if (!driveLink && rawBase64 && String(rawBase64).indexOf("data:image") === 0) {
          const saved = saveBase64PhotoToDrive(rawBase64, fileName, hariTanggal);
          if (saved) driveLink = saved.viewUrl;
        }
        const gpsInfo = a.lokasiAlamat ? (a.lokasiAlamat + (a.lokasiMapsUrl ? " (" + a.lokasiMapsUrl + ")" : "")) : (a.lokasiMapsUrl || "");
        return [a.id, hariTanggal, a.tanggal, a.waktuAbsen || "", a.karyawanId, a.namaKaryawan, a.pekerjaan || "", a.tipe, a.jamLembur, a.upahHarian, a.upahLembur, a.totalUpah, gpsInfo, driveLink ? fileName : "", driveLink, a.catatan || "", a.inputOleh || ""];
      });
      upsertRowsByKey(shAbs, headers, rows, 0);
    }

    if (Array.isArray(fullState.karyawan)) {
      const shEmp = ss.getSheetByName("Karyawan") || ss.insertSheet("Karyawan");
      const headers = ["ID_Karyawan", "Nama", "Jenis_Pekerjaan", "Alamat", "No_HP", "Gaji_Harian", "Lembur_Per_Jam", "Tim_Mandor"];
      const empNameMap = {};
      fullState.karyawan.forEach(function(item) {
        if (item && item.id) empNameMap[String(item.id)] = item.nama || item.id;
      });
      const rows = fullState.karyawan.map(function(k) {
        const subNames = Array.isArray(k.mandorOf)
          ? k.mandorOf.map(function(subId) { return empNameMap[String(subId)] || subId; }).join(", ")
          : "";
        return [k.id, k.nama, k.pekerjaan || "", k.alamat, k.hp, k.gajiHarian, k.lemburJam, subNames];
      });
      upsertRowsByKey(shEmp, headers, rows, 0);
    }

    if (Array.isArray(fullState.kasbon)) {
      const shKas = ss.getSheetByName("Kasbon") || ss.insertSheet("Kasbon");
      const headers = ["ID_Kasbon", "Tanggal", "ID_Karyawan", "Nama_Karyawan", "Nominal", "Keterangan", "Status"];
      const rows = fullState.kasbon.map(function(k) {
        return [k.id, k.tanggal, k.karyawanId, k.namaKaryawan, k.nominal, k.keterangan, k.status];
      });
      upsertRowsByKey(shKas, headers, rows, 0);
    }

    return ContentService.createTextOutput(JSON.stringify({ ok: true, spreadsheetId: ss.getId() }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: String(err) }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
